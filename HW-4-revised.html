TOTALNSA <- read.csv("TOTALNSA.csv")

TOTALNSA_ts <- ts(TOTALNSA$TOTALNSA,
                  start = c(1976, 1),
                  frequency = 12)

attributes(TOTALNSA_ts)
## $tsp
## [1] 1976.000 2026.667   12.000
## 
## $class
## [1] "ts"
plot(TOTALNSA_ts)

Acf(TOTALNSA_ts)

# Take the mean of all available history
mean_forecast <- meanf(TOTALNSA_ts, 5)
plot(mean_forecast)

# The mean forecast predicts future sales using the average of all historical observations.
# Naive
naive_forecast <- naive(TOTALNSA_ts, 5)
plot(naive_forecast)

# The naive forecast assumes sales over the next five months will remain at the most recent observed value.
# Random Walk
rwf_forecast <- rwf(TOTALNSA_ts, 5)
rwf_forecast <- rwf(TOTALNSA_ts, 5, drift=TRUE)
plot(rwf_forecast)

# This method uses the latest sales value and the average historical change to predict future sales.
# Seasonal Naive
snaive_forecast <- snaive(TOTALNSA_ts, 5)
plot(snaive_forecast)

# The seasonal naive method predicts sales using the same month from the previous year, which is useful for data that has seasonal patterns.
# Moving Averages
MA5_forecast <- ma(TOTALNSA_ts, order=5)
MA9_forecast <- ma(TOTALNSA_ts, order=9)

plot(TOTALNSA_ts)
lines(MA5_forecast, col="pink")
lines(MA9_forecast, col="blue")

# Moving averages smooth short-term fluctuations. The 9-month average is generally smoother than the 5-month average.
# Plot all models in a single chart
plot(mean_forecast)
lines(naive_forecast$mean, col="red")
lines(rwf_forecast$mean, col="green")
lines(snaive_forecast$mean, col="black")
lines(MA5_forecast, col="pink")
lines(MA9_forecast, col="blue")

# This chart displays the forecast predictions together. Moving averages mainly smooth historical data and are not forecasts of future sales in this example.

# Forecast

# These commands forecast vehicle sales for the next five months using the fitted ETS model.
# Split data into training and testing sets
n <- length(TOTALNSA_ts)

train <- head(TOTALNSA_ts, n - 5)
test <- tail(TOTALNSA_ts, 5)

# Fit forecasting models using training data
mean_test <- meanf(train, h=5)
naive_test <- naive(train, h=5)
rwf_test <- rwf(train, h=5, drift=TRUE)
snaive_test <- snaive(train, h=5)

ets_test_model <- ets(train)
ets_test <- forecast(ets_test_model, h=5)

HW_test_model <- HoltWinters(train)
## Warning in HoltWinters(train): optimization difficulties: ERROR:
## ABNORMAL_TERMINATION_IN_LNSRCH
HW_test <- forecast(HW_test_model, h=5)

# Compare test-set accuracy
rmse_results <- data.frame(
  Model = c(
    "Mean",
    "Naive",
    "Random Walk with Drift",
    "Seasonal Naive",
    "ETS",
    "Holt-Winters"
  ),
  RMSE = c(
    accuracy(mean_test, test)["Test set", "RMSE"],
    accuracy(naive_test, test)["Test set", "RMSE"],
    accuracy(rwf_test, test)["Test set", "RMSE"],
    accuracy(snaive_test, test)["Test set", "RMSE"],
    accuracy(ets_test, test)["Test set", "RMSE"],
    accuracy(HW_test, test)["Test set", "RMSE"]
  )
)

rmse_results <- rmse_results[
  order(rmse_results$RMSE),
]

print(rmse_results)

# The Holt-Winters model performed the best, with the lowest RMSE of 46.24, followed by ETS at 47.27, Naive at 55.15, Random Walk with Drift at 56.24, Seasonal Naive at 69.01, and Mean at 166.17.

# Therefore, Holt-Winters is the best forecasting model for this dataset because it produced the lowest RMSE, meaning its predictions were closest to the actual vehicle sales during the five-month test period.
